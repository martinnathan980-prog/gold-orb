function getToutLeContenu() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sBoites = ss.getSheetByName("1_BOITES");
  const sNom = ss.getSheetByName("2_NOMENCLATURE");
  if (!sBoites || !sNom) return { erreur: "Exécutez initialiserBaseDeDonnees() depuis l'éditeur." };
  const lireSheet = (sheet) => {
    const data = sheet.getDataRange().getValues();
    if (data.length < 2) return [];
    const headers = data[0].map(h => String(h).trim());
    const result = [];
    for (let i = 1; i < data.length; i++) {
      if (data[i].every(c => String(c).trim() === '')) continue;
      let obj = { _rowIndex: i + 1 };
      headers.forEach((h, j) => { obj[h] = data[i][j]; });
      result.push(obj);
    }
    return result;
  };
  return { boites: lireSheet(sBoites), nomenclature: lireSheet(sNom), headersBoites: sBoites.getRange(1,1,1,
sBoites.getLastColumn()).getValues()[0].map(h => String(h).trim()), headersNom: sNom.getRange(1,1,1,sNom.
getLastColumn()).getValues()[0].map(h => String(h).trim()) };
}
function saveLigne(sheetName, rowIndex, modifications) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName);
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
  for (const [key, value] of Object.entries(modifications)) {
    let colIndex = headers.indexOf(key);
    if (colIndex !== -1) sheet.getRange(rowIndex, colIndex + 1).setValue(value);
  }
  return getToutLeContenu();
}
function deleteLigne(sheetName, rowIndex) {
  SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName).deleteRow(rowIndex);
  return getToutLeContenu();
}
function deleteBoiteEntiere(pnGlobal, rowIndexBoite) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  ss.getSheetByName("1_BOITES").deleteRow(rowIndexBoite);
  const sNom = ss.getSheetByName("2_NOMENCLATURE");
  const dataNom = sNom.getDataRange().getValues();
  for (let i = dataNom.length - 1; i > 0; i--) {
    if (dataNom[i][1] === pnGlobal) sNom.deleteRow(i + 1);
  }
  return getToutLeContenu();
}
function addBoiteLibre(pnGlobal, fonction, dsVci) {
   const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("1_BOITES");
   const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
   let newRow = new Array(headers.length).fill('');
   headers.forEach((h, idx) => { if(h==="PN Global") newRow[idx]=pnGlobal; if(h==="Fonction")
newRow[idx]=fonction; if(h==="DS/VCI Associé") newRow[idx]=dsVci; if(h==="Statut") newRow[idx]="En étude";});
   sheet.appendRow(newRow); return getToutLeContenu();
}
function addSousEnsembleLibre(pnGlobal, type, pnType) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("2_NOMENCLATURE");
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
  let newRow = new Array(headers.length).fill('');
  headers.forEach((h, idx) => {
    if(h==="ID_Ligne") newRow[idx]="L-"+new Date().getTime(); if(h==="PN Global") newRow[idx]=pnGlobal;
    if(h==="Type") newRow[idx]=type; if(h==="PN du type") newRow[idx]=pnType;
  });
  sheet.appendRow(newRow); return getToutLeContenu();
}
function dupliquerBoite(pnSource, nouveauPn) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sBoites = ss.getSheetByName("1_BOITES");
  const sNom = ss.getSheetByName("2_NOMENCLATURE");
  const dataBoites = sBoites.getDataRange().getValues();
  let rowBoite = dataBoites.find(r => r[1] === pnSource);
  if(rowBoite) {
    let newRowB = [...rowBoite]; newRowB[1] = nouveauPn; newRowB[4] = "En étude";
    sBoites.appendRow(newRowB);
  }
  const dataNom = sNom.getDataRange().getValues();
  for(let i=1; i<dataNom.length; i++) {
    if(dataNom[i][1] === pnSource) {
      let newRowN = [...dataNom[i]]; newRowN[0] = "L-" + new Date().getTime() + Math.floor(Math.random()*1000);
newRowN[1] = nouveauPn;
      sNom.appendRow(newRowN);
    }
  }
  return getToutLeContenu();
}
function getCatalogueComposants() {
  try {
    const ss = SpreadsheetApp.openByUrl(URL_CATALOGUE);
    const data = ss.getSheets()[0].getDataRange().getValues();
    const headers = data[0].map(h => String(h).trim());
    const composants = [];
    for(let i=1; i<data.length; i++) {
      if(data[i].every(c => String(c).trim() === '')) continue;
      let obj = {}; headers.forEach((h, j) => { if(h) obj[h] = data[i][j]; }); composants.push(obj);
    }
    return composants;
  } catch(e) { return { error: "Erreur lecture catalogue" }; }
}

      function comparerBoite(pnSource) {
        const source = dataGlobale.boites.find(b => b['PN Global'] === pnSource);
        let resultats = [];
        dataGlobale.boites.forEach(cible => {
          if(cible['PN Global'] === pnSource) return;
          let score = 0; let tot = 0; let raisonsHTML = [];
          if(source['Fonction']) { tot+=15; if(cible['Fonction'] === source['Fonction']) { score+=15;
raisonsHTML.push(`<div class="analyze-row"><span class="analyze-icon">
</span><div class="analyze-
content"><b>Même Fonction:</b> ${source['Fonction']}</div></div>`); } else { raisonsHTML.push(`<div
class="analyze-row"><span class="analyze-icon text-danger">
</span><div class="analyze-content"><b>
Fonction:</b> ${source['Fonction']} vs ${cible['Fonction']}</div></div>`); } }
          if(source['Niveau de qualification']) { tot+=10; if(cible['Niveau de qualification'] === source['Niveau de
qualification']) { score+=10; raisonsHTML.push(`<div class="analyze-row"><span class="analyze-icon">
</span>
<div class="analyze-content"><b>Même Niveau Qualif:</b> ${source['Niveau de qualification']}</div></div>`); }
else { raisonsHTML.push(`<div class="analyze-row"><span class="analyze-icon text-danger">
</span><div
class="analyze-content"><b>Qualif. différente:</b> ${source['Niveau de qualification']} vs ${cible['Niveau de
qualification']||'-'}</div></div>`); } }
          let srcTypes = source.nomenclature.map(l=>l['Type']).filter(t=>t); let cibTypes = cible.nomenclature.map(l=>l['
Type']).filter(t=>t);
          if(srcTypes.length > 0) { tot+=25; let missing = srcTypes.filter(t => !cibTypes.includes(t)); if(missing.length
=== 0) { score+=25; raisonsHTML.push(`<div class="analyze-row"><span class="analyze-icon">
</span><div
class="analyze-content"><b>Architecture complète</b> (Possède tous les S/Ensembles requis)</div></div>`); }
else { raisonsHTML.push(`<div class="analyze-row"><span class="analyze-icon text-danger">
</span><div
class="analyze-content"><b>S/Ensembles manquants:</b> ${missing.join(', ')}</div></div>`); } }
          // ANALYSE DES COMPOSANTS STD GLOBAUX
          let srcStd = [...new Set(source.nomenclature.map(l=>l['Composant STD']).filter(t=>t).flatMap(s =>
s.split('\n').map(x=>x.trim())))];
          let cibStd = [...new Set(cible.nomenclature.map(l=>l['Composant STD']).filter(t=>t).flatMap(s =>
s.split('\n').map(x=>x.trim())))];
         
          if(srcStd.length > 0 || cibStd.length > 0) {
            tot+=50;
            let shared = srcStd.filter(t => cibStd.includes(t));
            let missingInCible = srcStd.filter(t => !cibStd.includes(t));
            let extraInCible = cibStd.filter(t => !srcStd.includes(t));
            if(srcStd.length > 0) { score += (50 * (shared.length/srcStd.length)); }
            let stdHtmlBlock = `<div class="mt-3">`;
            if(shared.length > 0) stdHtmlBlock += `<div class="p-2 mb-2 rounded" style="background:#f0fdf4;
border:1px solid #bbf7d0;"><div class="fw-bold text-success mb-1" style="font-size:11px; text-
transform:uppercase;">
 En commun (${shared.length})</div><div style="font-size:12px;
color:#166534;">${[...new Set(shared)].join('<br>')}</div></div>`;
            if(missingInCible.length > 0) stdHtmlBlock += `<div class="p-2 mb-2 rounded" style="background:#fffbeb;
border:1px solid #fef08a;"><div class="fw-bold mb-1" style="font-size:11px; text-transform:uppercase;
color:#b45309;">
 Absents de cet assemblage (${missingInCible.length})</div><div style="font-size:12px;
color:#92400e;">${[...new Set(missingInCible)].join('<br>')}</div></div>`;
            if(extraInCible.length > 0) stdHtmlBlock += `<div class="p-2 rounded" style="background:#f8fafc;
border:1px solid #e2e8f0;"><div class="fw-bold text-secondary mb-1" style="font-size:11px; text-
transform:uppercase;">
 Présents en plus (${extraInCible.length})</div><div style="font-size:12px;
color:#475569;">${[...new Set(extraInCible)].join('<br>')}</div></div>`;
            stdHtmlBlock += `</div>`;
            raisonsHTML.push(stdHtmlBlock);
          }
          let finalScore = tot > 0 ? Math.round((score/tot)*100) : 0;
          if(finalScore >= 15) resultats.push({ boite: cible, score: finalScore, html: raisonsHTML.join('') });
        });
        resultats.sort((a,b) => b.score - a.score);
        let finalHtml = resultats.length === 0 ? `<p class="text-center py-4">Aucune équivalence technique trouvée.
</p>` : resultats.map(res => `
          <div class="component-block">
            <div class="d-flex justify-content-between align-items-center mb-3 border-bottom pb-2">
              <div><h5 class="fw-bold text-primary m-0">${res.boite['PN Global']}</h5><div class="text-muted mt-1"
style="font-size:12px;">Fonction : <b>${res.boite['Fonction']}</b></div></div>
              <div class="fw-bold px-3 py-1" style="background:#f0f4f8; border-radius:8px;
color:#0071e3;">${res.score}% Match</div>
            </div>
            <div>${res.html}</div>
          </div>`).join('');
        document.getElementById('compareTitle').innerText = `Équivalences Globales : ${pnSource}`;
document.getElementById('compareResult').innerHTML = finalHtml; new bootstrap.Modal(document.
getElementById('compareModal')).show();
      }

            let extraInCible = cStd.filter(t => !sStd.includes(t));
            if(sStd.length > 0) { points += (35*(shared.length/sStd.length)); }
            let stdHtmlBlock = `<div class="mt-3">`;

transform:uppercase;">
 En commun (${shared.length})</div><div style="font-size:12px;

color:#b45309;">
 Absents de cette pièce (${missingInCible.length})</div><div style="font-size:12px;
color:#92400e;">${[...new Set(missingInCible)].join('<br>')}</div></div>`;
            if(extraInCible.length > 0) stdHtmlBlock += `<div class="p-2 rounded" style="background:#f8fafc;

transform:uppercase;">
 Présents en supplément (${extraInCible.length})</div><div style="font-size:12px;

            stdHtmlBlock += `</div>`;
            raisonsHTML.push(stdHtmlBlock);
          }
          if(totalPts===0) totalPts=1; let finalScore = Math.round((points/totalPts)*100);
         
          if(finalScore >= 15) {
             const parentBox = dataGlobale.boites.find(b => b['PN Global'] === cible['PN Global']);
             const fonctionTarget = parentBox ? parentBox['Fonction'] : 'Inconnue';
             resultats.push({ cible: cible, score: finalScore, fonction: fonctionTarget, html: raisonsHTML.join('') });
          }
        });

color:#0071e3;">${res.score}% Match</div>

          </div>
        `).join('');

(${sourceNom['PN du type']})`;
